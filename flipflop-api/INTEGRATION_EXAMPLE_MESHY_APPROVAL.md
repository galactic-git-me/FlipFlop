# Integration Example: Meshy Asset Approval Flow

## Complete flow showing how Meshy asset download integrates with the approval queue

---

## Scenario

Michael approves a 3D model generated by MeshyBot in the admin approval queue. The system should:
1. Download the `.glb` file from Meshy CDN
2. Store it locally in `data/uploads/3d-assets/`
3. Update the `Component3DAsset` record with the stable local URL
4. Mark the asset as approved and ready for use in the shop

---

## Step-by-Step Implementation

### 1. MeshyBot Submits Asset to Approval Queue

**MeshyBot creates approval queue item:**
```python
# In MeshyBot
approval_item = {
    "type": "model-3d",
    "subject_type": "case",
    "subject_id": 45,  # CaseCatalogue.id
    "category": "case",
    "meshy_task_id": "abc123",
    "meshy_glb_url": "https://cdn.meshy.ai/tasks/abc123/output.glb",
    "meshy_preview_url": "https://cdn.meshy.ai/tasks/abc123/preview.png",
    "poly_count": 50000,
    "file_size_kb": 2048,
    "asset_id": 123  # Component3DAsset.id (created earlier)
}

# Submit to bot_approval_queue
db.execute(
    "INSERT INTO bot_approval_queue (type, payload, created_at) VALUES (?, ?, ?)",
    ("model-3d", json.dumps(approval_item), datetime.utcnow())
)
```

**Admin UI calls approval endpoint:**

// flipflop-admin
async function approveAsset(approvalId: number, payload: any) {
  // 1. Mark approval queue item as approved
  await fetch(`/api/admin/approvals/${approvalId}/approve`, {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${adminToken}` }
  })
  
  // 2. Download and store Meshy asset locally
  const downloadResponse = await fetch('/api/admin/meshy-assets/download-from-approval', {
    method: 'POST',
    headers: { 
      'Authorization': `Bearer ${adminToken}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      asset_id: payload.asset_id,
      approval_payload: payload
    })
  })
  
  const result = await downloadResponse.json()

### 4. Backend Downloads and Stores Asset

**API endpoint handler:**
```python
# flipflop-api/app/api/meshy_asset_downloader.py

@router.post("/download-from-approval")
async def download_from_approval(
    request: ApprovalDownloadRequest,
    admin_email: str = Depends(get_current_admin),
    db: AsyncSession = Depends(get_db)
):
    try:
        base_dir = Path(__file__).resolve().parents[1] / "data" / "uploads"
        downloader = MeshyAssetDownloader(db, base_dir)
        
        # Download from Meshy and store locally
        glb_url, preview_url = await downloader.download_from_approval_queue(
            request.asset_id,
            request.approval_payload
        )
        
        # Update asset status to MESHY_DRAFT
        await downloader.update_asset_status_after_download(
            request.asset_id,
            Component3DAssetStatus.MESHY_DRAFT
        )
        
        log.info(
            "meshy_asset.approval_download_success",
            asset_id=request.asset_id,
            admin=admin_email
        )
        
        return {
            "success": True,
            "asset_id": request.asset_id,
            "glb_url": glb_url,
            "preview_url": preview_url,
            "message": "Asset downloaded from approval queue"
        }
        
    except Exception as e:
        log.error(
            "meshy_asset.approval_download_failed",
            asset_id=request.asset_id,
            error=str(e)
        )
        raise HTTPException(status_code=500, detail=f"Approval download failed: {str(e)}")
```

**Service downloads and stores:**
```python
# flipflop-api/app/services/meshy_asset_downloader.py

async def download_from_approval_queue(
    self,
    asset_id: int,
    approval_payload: dict
) -> Tuple[str, Optional[str]]:
    """
    Download asset from approval queue payload.
    """
    glb_url = approval_payload.get("meshy_glb_url")
    preview_url = approval_payload.get("meshy_preview_url")
    
    if not glb_url:
        raise ValueError("No meshy_glb_url in approval payload")
    
    # Download and store (internally calls download_and_store_asset)
    return await self.download_and_store_asset(
        asset_id,
        glb_url,
        preview_url
    )
```

**Behind the scenes:**
```python
async def download_and_store_asset(
    self,
    asset_id: int,
    meshy_glb_url: str,
    meshy_preview_url: Optional[str] = None,
    force_redownload: bool = False
) -> Tuple[str, Optional[str]]:
    # 1. Get asset record
    asset = await db.get(Component3DAsset, asset_id)
    
    # 2. Determine storage paths
    category = asset.category or "generic"
    filename_base = self._generate_filename(asset)
    # e.g., "case_123_case_v1_a3f5b2c8"
    
    # 3. Download .glb file
    glb_filename = f"{filename_base}.glb"
    glb_local_path = self.assets_dir / category / glb_filename
    # e.g., "data/uploads/3d-assets/case/case_123_case_v1_a3f5b2c8.glb"
    
    await self._download_file(meshy_glb_url, glb_local_path)
    
    # 4. Download preview if provided
    if meshy_preview_url:
        preview_filename = f"{filename_base}_preview.png"
        preview_local_path = self.assets_dir / category / preview_filename
        await self._download_file(meshy_preview_url, preview_local_path)
    
    # 5. Generate public URLs
    glb_public_url = f"/api/uploads/3d-assets/{category}/{glb_filename}"
    preview_public_url = f"/api/uploads/3d-assets/{category}/{preview_filename}" if meshy_preview_url else None
    
    # 6. Update asset record
    asset.glb_ref = glb_public_url
    asset.preview_image_ref = preview_public_url
    asset.file_size_kb = int(glb_local_path.stat().st_size / 1024)
    
    await db.commit()
    
    return glb_public_url, preview_public_url
```

## Complete Admin UI Flow (Future Enhancement)

### Approval Queue View

```typescript
// flipflop-admin/app/approvals/page.tsx

function ApprovalQueueItem({ item }: { item: ApprovalQueueItem }) {
  const [downloading, setDownloading] = useState(false)
  const [downloaded, setDownloaded] = useState(false)
  
  async function handleApprove() {
    setDownloading(true)
    
    try {
      // 1. Approve in queue
      await approveApprovalQueueItem(item.id)
      
      // 2. Download and store asset
      if (item.type === 'model-3d') {
        const result = await fetch('/api/admin/meshy-assets/download-from-approval', {
          method: 'POST',
          headers: { 
            'Authorization': `Bearer ${token}`,
            'Content-Type': 'application/json'
          },
          body: JSON.stringify({
            asset_id: item.payload.asset_id,
            approval_payload: item.payload
          })
        }).then(r => r.json())
        
        if (result.success) {
          setDownloaded(true)
          toast.success('Asset approved and stored locally')
        }
      }
    } catch (error) {
      toast.error('Failed to approve asset')
    } finally {
      setDownloading(false)
    }
  }
  
  return (
    <div className="approval-item">
      <div className="preview">
        {/* Use glb-proxy for preview before approval */}
        <model-viewer 
          src={`/glb-proxy?url=${encodeURIComponent(item.payload.meshy_glb_url)}`}
        />
      </div>
      
      <div className="actions">
        <button 
          onClick={handleApprove}
          disabled={downloading}
        >
          {downloading ? 'Downloading...' : 'Approve & Store'}
        </button>
        
        {downloaded && (
          <span className="success">✓ Stored locally</span>
        )}
      </div>
    </div>
  )
}
```

## Batch Approval Flow

For bulk approvals (e.g., after MeshyBot regenerates all 24 playbook assets):

```typescript
// flipflop-admin/app/approvals/page.tsx

async function batchApproveAll3DAssets() {
  // 1. Get all pending 3D assets
  const pending3DAssets = approvalQueue.filter(item => 
    item.type === 'model-3d' && item.status === 'pending'
  )
  
  // 2. Prepare batch download request
  const assets = pending3DAssets.map(item => ({
    asset_id: item.payload.asset_id,
    meshy_glb_url: item.payload.meshy_glb_url,
    meshy_preview_url: item.payload.meshy_preview_url
  }))
  
  // 3. Batch download and store
  const result = await fetch('/api/admin/meshy-assets/batch-download', {
    method: 'POST',
    headers: { 
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({ assets })
  }).then(r => r.json())
  
  // 4. Mark all as approved
  for (const item of pending3DAssets) {
    await approveApprovalQueueItem(item.id)
  }
  
  toast.success(`Approved and stored ${result.successful.length} assets`)
}
```

## Error Handling

### Meshy URL Expired During Approval

```typescript
try {
  const result = await fetch('/api/admin/meshy-assets/download-from-approval', {
    method: 'POST',
    body: JSON.stringify({ asset_id, approval_payload })
  }).then(r => r.json())
  
  if (!result.success) {
    // Meshy URL expired or download failed
    toast.error('Failed to download asset. Meshy URL may have expired.')
    
    // Notify MeshyBot to regenerate
    await fetch('/api/admin/meshy-bot/regenerate', {
      method: 'POST',
      body: JSON.stringify({ asset_id })
    })
    
    toast.info('Requested MeshyBot to regenerate asset')
  }
} catch (error) {
  console.error('Download error:', error)
}
```

### Download Timeout

```python
# In meshy_asset_downloader.py

async def _download_file(self, url: str, destination: Path):
    """Download file from URL to local path."""
    try:
        async with httpx.AsyncClient(timeout=120, follow_redirects=True) as client:
            response = await client.get(url)
            response.raise_for_status()
            
            destination.write_bytes(response.content)
            
    except httpx.TimeoutException:
        log.error("meshy_asset.download_timeout", url=url)
        raise RuntimeError(f"Download timed out after 120s: {url}")
        
    except httpx.HTTPError as e:
        log.error("meshy_asset.download_failed", url=url, error=str(e))
        raise RuntimeError(f"Failed to download {url}: {e}")
```
